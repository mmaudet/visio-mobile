> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a shared LayoutEngine in visio-core that sorts participants by voice activity with anti-flicker, manages pagination, and provides speaker mode logic.

**Architecture:** New `layout.rs` module in visio-core. The LayoutEngine is owned by RoomManager and driven by ActiveSpeakersChanged events. UIs receive pre-sorted participant lists and page info via new UniFFI methods/events. No layout computation happens on the native side.

**Tech Stack:** Rust (visio-core), UniFFI (UDL + codegen)

//! Layout engine: participant sorting, pagination, and speaker mode.

use crate::events::{ConnectionQuality, ParticipantInfo};
use std::time::{Duration, Instant};

/// Anti-flicker window: if two participants both spoke within this window,
/// sort them by identity (stable) instead of by exact last_spoke_at.
const ANTI_FLICKER_WINDOW: Duration = Duration::from_secs(10);

/// Sort participants by voice activity with anti-flicker.
pub fn sort_participants(participants: &mut [ParticipantInfo]) {
    let now = Instant::now();
    participants.sort_by(|a, b| {
        // 1. Hand raised first
        match (a.hand_raised, b.hand_raised) {
            (true, false) => return std::cmp::Ordering::Less,
            (false, true) => return std::cmp::Ordering::Greater,
            _ => {}
        }

        // 2. Voice activity with anti-flicker
        let a_spoke = a.last_spoke_at.map(|t| now.duration_since(t));
        let b_spoke = b.last_spoke_at.map(|t| now.duration_since(t));

        let a_in_window = a_spoke.map_or(false, |d| d < ANTI_FLICKER_WINDOW);
        let b_in_window = b_spoke.map_or(false, |d| d < ANTI_FLICKER_WINDOW);

        match (a_in_window, b_in_window) {
            (true, true) => {
                // Both spoke recently: sort by identity for stability
                a.identity.cmp(&b.identity)
            }
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (false, false) => {
                // Neither in window: sort by last_spoke_at descending
                match (a.last_spoke_at, b.last_spoke_at) {
                    (Some(a_t), Some(b_t)) => b_t.cmp(&a_t), // more recent first
                    (Some(_), None) => std::cmp::Ordering::Less,
                    (None, Some(_)) => std::cmp::Ordering::Greater,
                    (None, None) => std::cmp::Ordering::Equal,
                }
            }
        }
        // 3. Camera on before camera off
        .then_with(|| b.has_video.cmp(&a.has_video))
        // 4. Join time ascending
        .then_with(|| {
            match (a.joined_at, b.joined_at) {
                (Some(a_t), Some(b_t)) => a_t.cmp(&b_t),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => std::cmp::Ordering::Equal,
            }
        })
    });
}

use std::sync::Mutex;

pub enum LayoutMode {
    Grid,
    Speaker,
}

pub struct LayoutEngine {
    inner: Mutex<LayoutEngineInner>,
}

struct LayoutEngineInner {
    sorted_order: Vec<String>,
    layout_mode: LayoutMode,
    current_page: usize,
    page_size: usize,
    pinned_participant: Option<String>,
    main_speaker: Option<String>,
    main_speaker_since: Option<Instant>,
}

/// Anti-flicker for speaker mode main tile: don't switch if new speaker
/// has been talking for less than this duration.
const SPEAKER_ANTI_FLICKER: Duration = Duration::from_secs(3);

impl LayoutEngine {
    pub fn new() -> Self {
        Self {
            inner: Mutex::new(LayoutEngineInner {
                sorted_order: Vec::new(),
                layout_mode: LayoutMode::Grid,
                current_page: 0,
                page_size: 4,
                pinned_participant: None,
                main_speaker: None,
                main_speaker_since: None,
            }),
        }
    }

    pub fn update_sorted_order(&self, order: Vec<String>) {
        let mut inner = self.inner.lock().unwrap();
        inner.sorted_order = order;
        // Clamp current page if needed
        let max_page = inner.max_page();
        if inner.current_page > max_page {
            inner.current_page = max_page;
        }
    }

    pub fn set_page_size(&self, size: usize) {
        let mut inner = self.inner.lock().unwrap();
        inner.page_size = size.max(1);
        let max_page = inner.max_page();
        if inner.current_page > max_page {
            inner.current_page = max_page;
        }
    }

    pub fn set_current_page(&self, page: usize) {
        let mut inner = self.inner.lock().unwrap();
        let max_page = inner.max_page();
        inner.current_page = page.min(max_page);
    }

    pub fn current_page(&self) -> usize {
        self.inner.lock().unwrap().current_page
    }

    pub fn page_count(&self) -> usize {
        let inner = self.inner.lock().unwrap();
        if inner.sorted_order.is_empty() || inner.page_size == 0 {
            return 0;
        }
        (inner.sorted_order.len() + inner.page_size - 1) / inner.page_size
    }

    /// Returns SIDs for the current page.
    pub fn visible_participants(&self) -> Vec<String> {
        let inner = self.inner.lock().unwrap();
        inner.page_slice(inner.current_page)
    }

    /// Returns SIDs for pages ±1 (pre-cache targets).
    pub fn precached_participants(&self) -> Vec<String> {
        let inner = self.inner.lock().unwrap();
        let mut result = Vec::new();
        if inner.current_page > 0 {
            result.extend(inner.page_slice(inner.current_page - 1));
        }
        let next = inner.current_page + 1;
        if next < inner.page_count_inner() {
            result.extend(inner.page_slice(next));
        }
        result
    }

    pub fn set_layout_mode(&self, mode: LayoutMode) {
        self.inner.lock().unwrap().layout_mode = mode;
    }

    pub fn layout_mode(&self) -> bool {
        // Returns true if Speaker mode
        matches!(self.inner.lock().unwrap().layout_mode, LayoutMode::Speaker)
    }
}

impl LayoutEngineInner {
    fn max_page(&self) -> usize {
        let count = self.page_count_inner();
        if count == 0 { 0 } else { count - 1 }
    }

    fn page_count_inner(&self) -> usize {
        if self.sorted_order.is_empty() || self.page_size == 0 {
            return 0;
        }
        (self.sorted_order.len() + self.page_size - 1) / self.page_size
    }

    fn page_slice(&self, page: usize) -> Vec<String> {
        let start = page * self.page_size;
        let end = (start + self.page_size).min(self.sorted_order.len());
        if start >= self.sorted_order.len() {
            return Vec::new();
        }
        self.sorted_order[start..end].to_vec()
    }
}

    /// Update the main speaker based on active speakers list.
    /// `now` parameter allows deterministic testing.
    pub fn update_main_speaker(&self, active_speakers: &[String], now: Instant) {
        let mut inner = self.inner.lock().unwrap();
        let new_speaker = active_speakers.first().cloned();

        if let Some(ref new) = new_speaker {
            if let Some(ref current) = inner.main_speaker {
                if new != current {
                    // Anti-flicker: only switch if current speaker has held for > 3s
                    if let Some(since) = inner.main_speaker_since {
                        if now.duration_since(since) < SPEAKER_ANTI_FLICKER {
                            return; // Too soon, keep current
                        }
                    }
                }
            }
            inner.main_speaker = Some(new.clone());
            inner.main_speaker_since = Some(now);
        }
    }

    /// Returns the SID of the main participant (pinned > speaker).
    pub fn main_participant(&self) -> Option<String> {
        let inner = self.inner.lock().unwrap();
        inner.pinned_participant.clone().or_else(|| inner.main_speaker.clone())
    }

    /// Returns SIDs for the thumbnail strip (all except main).
    pub fn thumbnail_participants(&self) -> Vec<String> {
        let inner = self.inner.lock().unwrap();
        let main = inner.pinned_participant.as_ref().or(inner.main_speaker.as_ref());
        inner.sorted_order.iter()
            .filter(|sid| main.map_or(true, |m| *sid != m))
            .cloned()
            .collect()
    }

    /// Pin or unpin a participant as the main tile.
    pub fn pin_participant(&self, sid: Option<String>) {
        self.inner.lock().unwrap().pinned_participant = sid;
    }
