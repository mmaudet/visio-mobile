```rust
use aes_gcm::{Aes256Gcm, KeyInit, Nonce};
use aes_gcm::aead::Aead;
use base64::Engine;
use hkdf::Hkdf;
use sha2::Sha256;

const CHAT_ENCRYPTION_VERSION: u8 = 0x01;
const NONCE_SIZE: usize = 12;

/// Derive a 256-bit AES key from the room token using HKDF-SHA256.
fn derive_chat_key(room_token: &str) -> [u8; 32] {
    let hk = Hkdf::<Sha256>::new(Some(b"visio-chat-v1"), room_token.as_bytes());
    let mut key = [0u8; 32];
    hk.expand(b"aes-256-gcm", &mut key)
        .expect("32 bytes is a valid length for HKDF-SHA256");
    key
}

/// Encrypt a chat message. Returns base64-encoded ciphertext.
fn encrypt_message(plaintext: &str, key: &[u8; 32]) -> Result<String, VisioError> {
    let cipher = Aes256Gcm::new_from_slice(key)
        .map_err(|e| VisioError::Room(format!("cipher init: {e}")))?;

    let mut nonce_bytes = [0u8; NONCE_SIZE];
    rand::RngCore::fill_bytes(&mut rand::thread_rng(), &mut nonce_bytes);
    let nonce = Nonce::from_slice(&nonce_bytes);

    let ciphertext = cipher.encrypt(nonce, plaintext.as_bytes())
        .map_err(|e| VisioError::Room(format!("encrypt: {e}")))?;

    // [version][nonce][ciphertext+tag]
    let mut payload = Vec::with_capacity(1 + NONCE_SIZE + ciphertext.len());
    payload.push(CHAT_ENCRYPTION_VERSION);
    payload.extend_from_slice(&nonce_bytes);
    payload.extend_from_slice(&ciphertext);

    Ok(base64::engine::general_purpose::STANDARD.encode(&payload))
}

/// Decrypt a chat message. Input is base64-encoded.
fn decrypt_message(encoded: &str, key: &[u8; 32]) -> Result<String, VisioError> {
    let payload = base64::engine::general_purpose::STANDARD
        .decode(encoded)
        .map_err(|e| VisioError::Room(format!("base64 decode: {e}")))?;

    if payload.is_empty() {
        return Err(VisioError::Room("empty payload".into()));
    }

    let version = payload[0];
    if version != CHAT_ENCRYPTION_VERSION {
        return Err(VisioError::Room(format!("unknown encryption version: {version}")));
    }

    if payload.len() < 1 + NONCE_SIZE + 1 {
        return Err(VisioError::Room("payload too short".into()));
    }

    let nonce = Nonce::from_slice(&payload[1..1 + NONCE_SIZE]);
    let ciphertext = &payload[1 + NONCE_SIZE..];

    let cipher = Aes256Gcm::new_from_slice(key)
        .map_err(|e| VisioError::Room(format!("cipher init: {e}")))?;

    let plaintext = cipher.decrypt(nonce, ciphertext)
        .map_err(|_| VisioError::Room("decryption failed".into()))?;

    String::from_utf8(plaintext)
        .map_err(|e| VisioError::Room(format!("invalid UTF-8: {e}")))
}

/// Check if a message appears to be encrypted (starts with version byte after base64 decode).
fn is_encrypted_message(text: &str) -> bool {
    if let Ok(payload) = base64::engine::general_purpose::STANDARD.decode(text) {
        payload.first() == Some(&CHAT_ENCRYPTION_VERSION)
    } else {
        false
    }
}
```

```rust
//! Runtime feature flags via Unleash proxy.

use std::collections::HashMap;
use std::sync::{Arc, RwLock};
use std::time::Duration;

pub struct FeatureService {
    proxy_url: RwLock<Option<String>>,
    flags: Arc<RwLock<HashMap<String, bool>>>,
    defaults: HashMap<String, bool>,
}

impl FeatureService {
    pub fn new() -> Self {
        let defaults = HashMap::from([
            ("background_blur".into(), true),
            ("screen_share".into(), true),
            ("pip".into(), true),
            ("chat_encryption".into(), false),
            ("speaker_mode".into(), true),
            ("smart_subscriptions".into(), true),
            ("oidc_auth".into(), false),
        ]);

        Self {
            proxy_url: RwLock::new(None),
            flags: Arc::new(RwLock::new(HashMap::new())),
            defaults,
        }
    }

    /// Check if a feature is enabled. Checks runtime flags first, then defaults.
    /// Unknown features return false.
    pub fn is_enabled(&self, name: &str) -> bool {
        if let Some(&val) = self.flags.read().unwrap().get(name) {
            return val;
        }
        self.defaults.get(name).copied().unwrap_or(false)
    }

    /// Set the Unleash proxy URL. Pass None to disable remote flags.
    pub fn set_proxy_url(&self, url: Option<String>) {
        *self.proxy_url.write().unwrap() = url;
    }

    /// Fetch flags from the Unleash proxy. Non-blocking on failure.
    pub async fn refresh(&self) {
        let url = self.proxy_url.read().unwrap().clone();
        let Some(base_url) = url else { return };

        let client = match reqwest::Client::builder()
            .timeout(Duration::from_secs(3))
            .build()
        {
            Ok(c) => c,
            Err(e) => {
                tracing::warn!("feature flags: failed to build client: {e}");
                return;
            }
        };

        let endpoint = format!("{base_url}/api/client/features");
        let resp = match client.get(&endpoint).send().await {
            Ok(r) => r,
            Err(e) => {
                tracing::warn!("feature flags: fetch failed: {e}");
                return;
            }
        };

        let body: serde_json::Value = match resp.json().await {
            Ok(v) => v,
            Err(e) => {
                tracing::warn!("feature flags: parse failed: {e}");
                return;
            }
        };

        if let Some(toggles) = body.get("toggles").and_then(|t| t.as_array()) {
            let mut flags = self.flags.write().unwrap();
            for toggle in toggles {
                if let (Some(name), Some(enabled)) = (
                    toggle.get("name").and_then(|n| n.as_str()),
                    toggle.get("enabled").and_then(|e| e.as_bool()),
                ) {
                    flags.insert(name.to_string(), enabled);
                }
            }
        }
    }

    /// Start periodic refresh (every 60s). Call from a tokio task.
    pub async fn start_periodic_refresh(self: Arc<Self>) {
        let mut interval = tokio::time::interval(Duration::from_secs(60));
        loop {
            interval.tick().await;
            self.refresh().await;
        }
    }
}
```

```bash
#!/bin/bash
set -euo pipefail

# Check for missing i18n keys across all locales.
# Usage: ./scripts/check-i18n.sh
# Exit 0 if all keys present, 1 if any missing.

REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
I18N_DIR="$REPO_ROOT/i18n"
SOURCE="$I18N_DIR/en.json"

if [ ! -f "$SOURCE" ]; then
    echo "ERROR: Source file $SOURCE not found"
    exit 1
fi

# Extract keys from en.json
SOURCE_KEYS=$(python3 -c "
import json, sys
with open('$SOURCE') as f:
    data = json.load(f)
for key in sorted(data.keys()):
    print(key)
")

MISSING=0
for locale_file in "$I18N_DIR"/*.json; do
    locale=$(basename "$locale_file" .json)
    [ "$locale" = "en" ] && continue

    LOCALE_KEYS=$(python3 -c "
import json, sys
with open('$locale_file') as f:
    data = json.load(f)
for key in sorted(data.keys()):
    print(key)
")

    DIFF=$(comm -23 <(echo "$SOURCE_KEYS") <(echo "$LOCALE_KEYS"))
    if [ -n "$DIFF" ]; then
        echo "MISSING in $locale.json:"
        echo "$DIFF" | sed 's/^/  - /'
        MISSING=1
    fi
done

if [ "$MISSING" -eq 0 ]; then
    echo "All locales have all keys from en.json"
fi
exit $MISSING
```
