## Task 3: iOS — A2DP → HFP transition gap handling

**Files:**
- Modify: `ios/VisioMobile/Services/ContextDetector.swift`

- [ ] **Step 1: Detect HFP transition gap in ContextDetector**

In `ContextDetector.swift`, modify `handleAudioRouteChange()` (lines 94-127) to detect the transition gap. The gap is when a BT device is known to be connected but no BT port appears in the current route:

```swift
private var knownBluetoothConnected = false

private func handleAudioRouteChange(_ notification: Notification) {
    guard let userInfo = notification.userInfo,
          let reasonValue = userInfo[AVAudioSessionRouteChangeReasonKey] as? UInt,
          let reason = AVAudioSession.RouteChangeReason(rawValue: reasonValue) else { return }

    let currentRoute = AVAudioSession.sharedInstance().currentRoute
    let hasBtOutput = currentRoute.outputs.contains { port in
        [.bluetoothA2DP, .bluetoothHFP, .bluetoothLE, .carAudio].contains(port.portType)
    }

    if hasBtOutput {
        knownBluetoothConnected = true
        manager?.routeAudioToBluetooth()
        reportBluetoothCarKit(currentRoute: currentRoute)
    } else if knownBluetoothConnected && !hasBtOutput {
        // A2DP → HFP gap: BT was connected but no BT port in route
        // This is a transient state during profile renegotiation
        // Fallback to speaker temporarily — HFP will reconnect in ~200-500ms
        if reason == .routeConfigurationChange || reason == .categoryChange {
            // Transient gap — do NOT clear knownBluetoothConnected
            // The next route change with BT port will restore routing
            return
        }
        // Actual disconnect
        knownBluetoothConnected = false
        manager?.restoreDefaultAudioRoute()
        manager?.reportBluetoothCarKit(connected: false)
    }
}
```

```rust
/// Pre-check: verify the LiveKit server is reachable before attempting
/// a full connection. Converts wss:// to https:// for the HEAD request.
pub async fn prepare_connection(&self, livekit_url: &str) -> Result<(), VisioError> {
    let http_url = livekit_url
        .replace("wss://", "https://")
        .replace("ws://", "http://");

    let client = reqwest::Client::builder()
        .timeout(std::time::Duration::from_secs(5))
        .build()
        .map_err(|e| VisioError::NetworkUnreachable(e.to_string()))?;

    client
        .head(&http_url)
        .send()
        .await
        .map_err(|e| VisioError::NetworkUnreachable(format!("{http_url}: {e}")))?;

    Ok(())
}
```

```rust
struct TokenCache {
    inner: std::sync::Mutex<Option<(String, String, Instant)>>,
}

impl TokenCache {
    fn new() -> Self {
        Self { inner: std::sync::Mutex::new(None) }
    }

    fn store(&self, token: String, url: String) {
        *self.inner.lock().unwrap() = Some((token, url, Instant::now()));
    }

    fn get(&self, max_age: std::time::Duration) -> Option<(String, String)> {
        let guard = self.inner.lock().unwrap();
        guard.as_ref().and_then(|(token, url, stored_at)| {
            if stored_at.elapsed() < max_age {
                Some((token.clone(), url.clone()))
            } else {
                None
            }
        })
    }

    fn clear(&self) {
        *self.inner.lock().unwrap() = None;
    }
}
```

```rust
/// Compare current participant list with the room's actual state.
/// Returns (joined, left) SIDs for participants that are out of sync.
pub fn resync(&mut self, room_participants: Vec<ParticipantInfo>) -> (Vec<ParticipantInfo>, Vec<String>) {
    let current_sids: std::collections::HashSet<_> = self.participants.iter().map(|p| p.sid.clone()).collect();
    let room_sids: std::collections::HashSet<_> = room_participants.iter().map(|p| p.sid.clone()).collect();

    let joined: Vec<ParticipantInfo> = room_participants
        .into_iter()
        .filter(|p| !current_sids.contains(&p.sid))
        .collect();

    let left: Vec<String> = current_sids
        .into_iter()
        .filter(|sid| !room_sids.contains(sid))
        .collect();

    // Apply changes
    for p in &joined {
        self.add_participant(p.clone());
    }
    for sid in &left {
        self.remove_participant(sid);
    }

    (joined, left)
}
```

```rust
pub trait InitProgressListener: Send + Sync {
    fn on_phase_started(&self, phase: InitPhase);
    fn on_phase_completed(&self, phase: InitPhase, result: InitResult, error: Option<InitPhaseError>);
}

/// No-op listener for the legacy constructor that doesn't care about phases.
pub struct NoOpListener;
impl InitProgressListener for NoOpListener {
    fn on_phase_started(&self, _phase: InitPhase) {}
    fn on_phase_completed(&self, _phase: InitPhase, _result: InitResult, _error: Option<InitPhaseError>) {}
}

pub struct InitSequence {
    listener: Box<dyn InitProgressListener>,
}

impl InitSequence {
    pub fn new(listener: Box<dyn InitProgressListener>) -> Self {
        Self { listener }
    }

    /// Phase 1: Load settings from disk. Always succeeds unless config is corrupted.
    pub fn init_settings(&self, data_dir: &str) -> Result<Arc<SettingsStore>, InitPhaseError> {
        self.listener.on_phase_started(InitPhase::Settings);
        let store = Arc::new(SettingsStore::new(data_dir));
        self.listener.on_phase_completed(InitPhase::Settings, InitResult::Success, None);
        Ok(store)
    }

    /// Phase 2: Restore authentication session. May fail (token expired, no session).
    pub async fn init_auth(
        &self,
        session_manager: &std::sync::Mutex<SessionManager>,
        settings: &SettingsStore,
    ) -> InitResult {
        self.listener.on_phase_started(InitPhase::Auth);

        // Try to restore session from each configured instance
        let instances = settings.get_meet_instances();
        let mut any_restored = false;

        for instance in &instances {
            let sm = session_manager.lock().unwrap();
            if sm.has_session(instance) {
                any_restored = true;
                break;
            }
        }

        let result = InitResult::Success;
        self.listener.on_phase_completed(InitPhase::Auth, result.clone(), None);
        result
    }

    /// Phase 3: Start optional services. Each may fail independently.
    pub async fn init_services(&self) -> InitResult {
        self.listener.on_phase_started(InitPhase::Services);
        // Calendar sync and feature flags will be started here.
        // For now, this phase is a no-op placeholder that succeeds.
        let result = InitResult::Success;
        self.listener.on_phase_completed(InitPhase::Services, result.clone(), None);
        result
    }

    /// Phase 4: Mark as ready.
    pub fn ready(&self) {
        self.listener.on_phase_started(InitPhase::Ready);
        self.listener.on_phase_completed(InitPhase::Ready, InitResult::Success, None);
    }
}
```

```rust
pub fn new_with_listener(data_dir: String, listener: Box<dyn visio_core::InitProgressListener>) -> Self {
    visio_log("VISIO FFI: VisioClient::new_with_listener() called");
    let rt = tokio::runtime::Runtime::new().expect("failed to create tokio runtime");

    let init = visio_core::InitSequence::new(listener);

    // Phase 1: Settings
    let settings = init.init_settings(&data_dir)
        .expect("settings initialization failed");

    let room_manager = visio_core::RoomManager::new();

    // Platform-specific playout buffer setup (same as existing new())
    #[cfg(target_os = "android")]
    {
        let buf = room_manager.playout_buffer();
        *PLAYOUT_BUFFER.lock().unwrap() = Some(buf);
    }
    #[cfg(target_os = "ios")]
    {
        let buf = room_manager.playout_buffer();
        *PLAYOUT_BUFFER_IOS.lock().unwrap() = Some(buf);
    }

    let controls = room_manager.controls();
    let chat = room_manager.chat();
    let session_manager = Arc::new(StdMutex::new(visio_core::SessionManager::new()));

    // Phase 2: Auth
    rt.block_on(init.init_auth(&session_manager, &settings));

    let calendar = Arc::new(visio_core::CalendarService::new(
        settings.clone(),
        room_manager.emitter(),
        data_dir.clone(),
    ));

    // Phase 3: Services (calendar, future feature flags)
    rt.block_on(init.init_services());

    if settings.get_calendar_url().is_some() {
        let cal = calendar.clone();
        rt.spawn(async move { cal.refresh().await; });
        let cal = calendar.clone();
        rt.spawn(async move { let _ = cal.start_periodic_refresh().await; });
    }

    // Phase 4: Ready
    init.ready();

    Self {
        room_manager,
        controls,
        chat,
        settings,
        session_manager,
        calendar,
        rt,
    }
}
```
