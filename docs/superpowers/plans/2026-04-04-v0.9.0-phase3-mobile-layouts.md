# v0.9.0 Phase 3 — Layouts Implementation Plan (Mobile + Desktop)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the flat video grid with a paginated swipeable layout (mobile) and add speaker mode + voice-activity sorting on all three platforms (Android, iOS, Desktop).

**Architecture:** UIs are renderers — they call the core's LayoutEngine (Phase 2) for sorted participant lists, page info, and speaker state. All layout decisions are in Rust. The Android/iOS code handles only visual rendering and user gesture forwarding.

**Tech Stack:** Kotlin + Jetpack Compose (Android), Swift + SwiftUI (iOS), React + TypeScript (Desktop/Tauri), UniFFI bindings from Phase 2, Tauri invoke commands

**Spec:** `docs/superpowers/specs/2026-04-04-v0.9.0-release-design.md` — Phase 3 section

**Prerequisite:** Phase 2 must be merged. The LayoutEngine API (`setPageSize`, `setCurrentPage`, `visibleParticipants`, `mainParticipant`, `thumbnailParticipants`, etc.) must be available in the generated UniFFI bindings.

```kotlin
@Composable
fun PaginatedGridLayout(
    participants: List<ParticipantInfo>,
    activeSpeakers: List<String>,
    client: VisioClient,
    onTileTap: (DisplayItem) -> Unit,
    onTileLongPress: (DisplayItem) -> Unit,
    modifier: Modifier = Modifier,
) {
    BoxWithConstraints(modifier = modifier) {
        val isLandscape = maxWidth > maxHeight
        val columns = when {
            participants.size <= 1 -> 1
            isLandscape -> 3
            else -> 2
        }
        val rows = if (isLandscape) {
            if (maxHeight > 600.dp) 2 else 1
        } else {
            2
        }
        val pageSize = columns * rows

        // Inform the core of the page size
        LaunchedEffect(pageSize) {
            client.setPageSize(pageSize.toUInt())
        }

        val pageCount = client.pageCount().toInt().coerceAtLeast(1)
        val pagerState = rememberPagerState(pageCount = { pageCount })

        // Sync pager state → core
        LaunchedEffect(pagerState.currentPage) {
            client.setCurrentPage(pagerState.currentPage.toUInt())
        }

        Column {
            HorizontalPager(
                state = pagerState,
                modifier = Modifier.weight(1f),
            ) { page ->
                // Get visible participants for this page from core
                val visibleSids = if (page == pagerState.currentPage) {
                    client.visibleParticipants()
                } else {
                    // For adjacent pages, use precached list or compute
                    client.precachedParticipants()
                        .drop((page - pagerState.currentPage.coerceAtLeast(1)) * pageSize)
                        .take(pageSize)
                }

                val pageParticipants = participants.filter { p -> visibleSids.contains(p.sid) }
                val displayItems = buildDisplayItems(pageParticipants)

                GridPage(
                    displayItems = displayItems,
                    columns = columns,
                    rows = rows,
                    activeSpeakers = activeSpeakers,
                    onTileTap = onTileTap,
                    onTileLongPress = onTileLongPress,
                    maxHeight = maxHeight,
                    spacing = 8.dp,
                )
            }

            // Page indicator
            if (pageCount > 1) {
                PageIndicator(
                    currentPage = pagerState.currentPage,
                    totalPages = pageCount,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 4.dp),
                )
            }
        }
    }
}

@Composable
fun GridPage(
    displayItems: List<DisplayItem>,
    columns: Int,
    rows: Int,
    activeSpeakers: List<String>,
    onTileTap: (DisplayItem) -> Unit,
    onTileLongPress: (DisplayItem) -> Unit,
    maxHeight: Dp,
    spacing: Dp,
) {
    val rowCount = (displayItems.size + columns - 1) / columns
    val rowHeight = (maxHeight - spacing * (rowCount - 1).coerceAtLeast(0)) / rowCount.coerceAtLeast(1)

    Column(
        verticalArrangement = Arrangement.spacedBy(spacing),
        modifier = Modifier.fillMaxSize().padding(horizontal = spacing),
    ) {
        for (row in 0 until rowCount) {
            Row(
                horizontalArrangement = Arrangement.spacedBy(spacing),
                modifier = Modifier.height(rowHeight),
            ) {
                for (col in 0 until columns) {
                    val index = row * columns + col
                    if (index < displayItems.size) {
                        val item = displayItems[index]
                        VideoTile(
                            item = item,
                            activeSpeakers = activeSpeakers,
                            onTap = { onTileTap(item) },
                            onLongPress = { onTileLongPress(item) },
                            modifier = Modifier.weight(1f),
                        )
                    } else {
                        Spacer(modifier = Modifier.weight(1f))
                    }
                }
            }
        }
    }
}

@Composable
fun PageIndicator(currentPage: Int, totalPages: Int, modifier: Modifier = Modifier) {
    if (totalPages <= 8) {
        // Dots indicator
        Row(
            horizontalArrangement = Arrangement.Center,
            modifier = modifier,
        ) {
            repeat(totalPages) { index ->
                Box(
                    modifier = Modifier
                        .padding(horizontal = 3.dp)
                        .size(if (index == currentPage) 8.dp else 6.dp)
                        .background(
                            color = if (index == currentPage) MaterialTheme.colorScheme.primary
                                    else MaterialTheme.colorScheme.onSurface.copy(alpha = 0.3f),
                            shape = CircleShape,
                        )
                )
            }
        }
    } else {
        // Compact text indicator
        Text(
            text = "${currentPage + 1}/$totalPages",
            style = MaterialTheme.typography.labelSmall,
            color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.7f),
            modifier = modifier.then(Modifier.fillMaxWidth()),
            textAlign = TextAlign.Center,
        )
    }
}

@Composable
fun SpeakerLayout(
    mainSid: String?,
    thumbnails: List<String>,
    participants: List<ParticipantInfo>,
    activeSpeakers: List<String>,
    onThumbnailTap: (String) -> Unit,
    modifier: Modifier = Modifier,
) {
    val participantMap = remember(participants) { participants.associateBy { it.sid } }

    Column(modifier = modifier.fillMaxSize()) {
        // Main tile: 70% of height
        Box(modifier = Modifier.weight(0.7f).fillMaxWidth().padding(8.dp)) {
            val mainParticipant = mainSid?.let { participantMap[it] }
            if (mainParticipant != null) {
                val item = buildDisplayItems(listOf(mainParticipant)).firstOrNull()
                if (item != null) {
                    VideoTile(
                        item = item,
                        activeSpeakers = activeSpeakers,
                        onTap = {},
                        onLongPress = {},
                        modifier = Modifier.fillMaxSize(),
                    )
                }
            }
        }

        // Thumbnail strip: ~100dp
        LazyRow(
            horizontalArrangement = Arrangement.spacedBy(4.dp),
            contentPadding = PaddingValues(horizontal = 8.dp),
            modifier = Modifier.height(100.dp).fillMaxWidth(),
        ) {
            items(thumbnails.size) { index ->
                val sid = thumbnails[index]
                val participant = participantMap[sid]
                if (participant != null) {
                    val item = buildDisplayItems(listOf(participant)).firstOrNull()
                    if (item != null) {
                        val isPinned = mainSid == sid
                        VideoTile(
                            item = item,
                            activeSpeakers = activeSpeakers,
                            onTap = { onThumbnailTap(sid) },
                            onLongPress = {},
                            modifier = Modifier
                                .width(133.dp) // 4:3 ratio with 100dp height
                                .fillMaxHeight()
                                .then(
                                    if (isPinned) Modifier.border(2.dp, MaterialTheme.colorScheme.primary, RoundedCornerShape(4.dp))
                                    else Modifier
                                ),
                        )
                    }
                }
            }
        }
    }
}

private var paginatedGridLayout: some View {
    GeometryReader { geo in
        let isLandscape = geo.size.width > geo.size.height
        let columns = participants.count <= 1 ? 1 : (isLandscape ? 3 : 2)
        let rows = isLandscape ? (geo.size.height > 500 ? 2 : 1) : 2
        let pageSize = columns * rows

        let _ = updatePageSize(pageSize)
        let pageCount = max(Int(client.pageCount()), 1)

        TabView(selection: $currentPage) {
            ForEach(0..<pageCount, id: \.self) { page in
                gridPage(
                    page: page,
                    columns: columns,
                    rows: rows,
                    geo: geo
                )
                .tag(page)
            }
        }
        .tabViewStyle(.page(indexDisplayMode: pageCount <= 8 ? .automatic : .never))
        .overlay(alignment: .bottom) {
            if pageCount > 8 {
                Text("\(currentPage + 1)/\(pageCount)")
                    .font(.caption2)
                    .foregroundColor(.secondary)
                    .padding(.bottom, 4)
            }
        }
        .onChange(of: currentPage) { newPage in
            client.setCurrentPage(UInt32(newPage))
        }
    }
}

@State private var currentPage = 0

private func updatePageSize(_ size: Int) {
    client.setPageSize(UInt32(size))
}

private func gridPage(page: Int, columns: Int, rows: Int, geo: GeometryProxy) -> some View {
    let visibleSids = client.visibleParticipants()
    let pageParticipants = participants.filter { visibleSids.contains($0.sid) }
    let displayItems = buildDisplayItems(from: pageParticipants)
    let spacing: CGFloat = 8
    let rowCount = max(1, (displayItems.count + columns - 1) / columns)
    let rowHeight = (geo.size.height - spacing * CGFloat(rowCount - 1)) / CGFloat(rowCount)

    return VStack(spacing: spacing) {
        ForEach(0..<rowCount, id: \.self) { row in
            HStack(spacing: spacing) {
                ForEach(0..<columns, id: \.self) { col in
                    let index = row * columns + col
                    if index < displayItems.count {
                        let item = displayItems[index]
                        VideoTileView(item: item, activeSpeakers: activeSpeakers)
                            .onTapGesture { focusItem(item) }
                            .onLongPressGesture { pinItem(item) }
                    } else {
                        Color.clear
                    }
                }
            }
            .frame(height: rowHeight)
        }
    }
    .padding(.horizontal, spacing)
}
```

// SpeakerLayout.tsx
import { useEffect, useState } from 'react';
import { invoke } from '@tauri-apps/api/core';

interface SpeakerLayoutProps {
  participants: ParticipantInfo[];
  activeSpeakers: string[];
}

export function SpeakerLayout({ participants, activeSpeakers }: SpeakerLayoutProps) {
  const [mainSid, setMainSid] = useState<string | null>(null);
  const [thumbnails, setThumbnails] = useState<string[]>([]);

  useEffect(() => {
    async function fetchLayout() {
      const main = await invoke<string | null>('main_participant');
      const thumbs = await invoke<string[]>('thumbnail_participants');
      setMainSid(main);
      setThumbnails(thumbs);
    }
    fetchLayout();
  }, [participants, activeSpeakers]);

  const mainParticipant = participants.find(p => p.sid === mainSid);
  const participantMap = Object.fromEntries(participants.map(p => [p.sid, p]));

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%' }}>
      {/* Main tile: 70% height */}
      <div style={{ flex: '0 0 70%', padding: 8 }}>
        {mainParticipant && <VideoTile participant={mainParticipant} large />}
      </div>

      {/* Thumbnail strip */}
      <div style={{
        display: 'flex',
        gap: 4,
        overflowX: 'auto',
        padding: '0 8px',
        height: 100,
      }}>
        {thumbnails.map(sid => {
          const p = participantMap[sid];
          if (!p) return null;
          return (
            <div
              key={sid}
              style={{
                width: 133,
                height: 100,
                flexShrink: 0,
                border: sid === mainSid ? '2px solid var(--accent)' : 'none',
                borderRadius: 4,
                cursor: 'pointer',
              }}
              onClick={() => invoke('pin_participant', { sid })}
            >
              <VideoTile participant={p} />
            </div>
          );
        })}
      </div>
    </div>
  );
}
